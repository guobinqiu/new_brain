# 在项目根目录执行以下命令

# ModelScope 向量模型
modelscope download BAAI/bge-base-zh-v1.5 --local-dir models/BAAI/bge-base-zh-v1.5
modelscope download BAAI/bge-base-en-v1.5 --local-dir models/BAAI/bge-base-en-v1.5
modelscope download BAAI/bge-m3 --local-dir models/BAAI/bge-m3

# ModelScope 重排模型
modelscope download BAAI/bge-reranker-base --local-dir models/BAAI/bge-reranker-base
modelscope download BAAI/bge-reranker-v2-m3 --local-dir models/BAAI/bge-reranker-v2-m3
modelscope download BAAI/bge-reranker-large --local-dir models/BAAI/bge-reranker-large

# MinerU 4.0 模型；使用 Parser 环境中的 mineru-kit，与运行时共用模型根目录
export MINERU_MODEL_BASE_DIR="$PWD/models/mineru"

# basic：仅小模型；后端自动按当前机器选择
uv run --project services/parser --python 3.11 mineru-kit models download --tier basic --source modelscope
uv run --project services/parser --python 3.11 mineru-kit models verify --tier basic

# standard/advanced：小模型 + VLM；以下按部署目标二选一
# CPU：运行时同样设置 MINERU_MODEL_SMALL_BACKEND=onnx、MINERU_MODEL_VLM_ENGINE=llama-cpp
uv run --project services/parser --python 3.11 mineru-kit models download --tier standard --small-backend onnx --vlm-engine llama-cpp --source modelscope
uv run --project services/parser --python 3.11 mineru-kit models verify --tier standard --small-backend onnx --vlm-engine llama-cpp

# NVIDIA GPU：运行时同样设置 MINERU_MODEL_SMALL_BACKEND=torch、MINERU_MODEL_VLM_ENGINE=vllm
uv run --project services/parser --python 3.11 --extra gpu mineru-kit models download --tier standard --small-backend torch --vlm-engine vllm --source modelscope
uv run --project services/parser --python 3.11 --extra gpu mineru-kit models verify --tier standard --small-backend torch --vlm-engine vllm
